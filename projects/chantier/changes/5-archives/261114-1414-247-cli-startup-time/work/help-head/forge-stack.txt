usage: chantier forge-stack [-h] [--batch BATCH] [--stack N]
                            {create,show,rebuild}

positional arguments:
  {create,show,rebuild}

options:
  -h, --help            show this help message and exit
  --batch BATCH         batch number, 4 or batch-04 (default: $CHANTIER_BATCH,
                        else the only running one)
  --stack N             show: the stack's number (default: the one that holds
                        the batch's PRs)

exit=0
