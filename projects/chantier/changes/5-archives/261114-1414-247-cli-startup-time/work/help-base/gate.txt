usage: chantier gate [-h] [--batch BATCH] [--stop] cr [command ...]

positional arguments:
  cr
  command        the long gate's commands, from the project's rules: one per
                 argument, never chained with ; or &&; they stop at the first
                 failure

options:
  -h, --help     show this help message and exit
  --batch BATCH  batch number, 4 or batch-04 (default: $CHANTIER_BATCH, else
                 the only running one)
  --stop         stop this CR's running long gate: its suite, the gate lock;
                 recorded `stopped`, no pass; nothing else changes

exit=0
