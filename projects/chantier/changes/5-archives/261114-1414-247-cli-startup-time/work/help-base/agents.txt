usage: chantier agents [-h] [--batch BATCH] [--epoch EPOCH] {set} cr agent

positional arguments:
  {set}
  cr
  agent          the agent id; `-` clears it

options:
  -h, --help     show this help message and exit
  --batch BATCH  batch number, 4 or batch-04 (default: $CHANTIER_BATCH, else
                 the only running one)
  --epoch EPOCH  the Foreman's epoch; stale ones exit 3

exit=0
