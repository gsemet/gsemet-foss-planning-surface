const LONG = "x".repeat(150);
const code = ["```text", LONG, "```"];
const table = ["| First column | Second column |", "| --- | --- |", `| \`${"a".repeat(70)}\` | \`${"b".repeat(70)}\` |`];
const indent = (lines: string[], by: string): string[] => lines.map((l) => by + l);
const CASES: Array<[string, string[]]> = [
  ["top-code", code],
  ["top-table", table],
  ["top-image", ["![w](wide.svg)"]],
  ["linked-image", ["[![w](wide.svg)](https://example.com)"]],
  ["emphasised-image", ["*![w](wide.svg)*"]],
  ["narrow-then-wide", ["![n](narrow.svg)", "![w](wide.svg)"]],
  ["wide-then-narrow", ["![w](wide.svg)", "![n](narrow.svg)"]],
  ["linked-wide-then-plain-narrow", ["[![w](wide.svg)](https://example.com)", "![n](narrow.svg)"]],
  ["list-code", ["- item", "", ...indent(code, "  ")]],
  ["list-table", ["- item", "", ...indent(table, "  ")]],
  ["admonition-table", ["```{note}", ...table, "```"]],
  ["admonition-code", ["````{note}", ...code, "````"]],
  ["quote-image", ["> ![w](wide.svg)"]],
  ["quote-code", indent(code, "> ")],
  ["card-code", ["````{card} Card", ...code, "````"]],
  ["card-table", ["```{card} Card", ...table, "```"]],
  ["tabset-code", ["`````{tab-set}", "````{tab-item} A", ...code, "````", "`````"]],
  ["tabset-image", ["````{tab-set}", "```{tab-item} A", "![w](wide.svg)", "```", "````"]],
  ["dropdown-code", ["````{dropdown} Drop", ":open:", "", ...code, "````"]],
  ["dropdown-table", ["```{dropdown} Drop", ":open:", "", ...table, "```"]],
];

async function read(page: Page): Promise<Record<string, string[]>> {
  await page.evaluate(async () => {
    await document.fonts.ready;
    await Promise.all(Array.from(document.images).map((i) => i.decode().catch(() => undefined)));
    window.dispatchEvent(new Event("resize"));
    await new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(resolve)));
  });
  return page.evaluate(() => {
    const content = document.getElementById("sphinx-myst-preview-content")!;
    const out: Record<string, string[]> = {};
    let name = "";
    for (const child of Array.from(content.children)) {
      if (child.tagName === "H2") { name = (child.textContent ?? "").trim().split(/\s+/u)[0]!.replace(/[^a-z-]/gu, ""); out[name] = []; continue; }
      if (!name) continue;
      for (const el of [child, ...Array.from(child.querySelectorAll("*"))]) {
        const interesting = el === child || el.matches("table, figure, img, pre, li, details, blockquote, p, .myst-tab-set, .myst-tab-panel")
          || el.hasAttribute("data-sphinx-preview-wide");
        if (!interesting) continue;
        const box = el.getBoundingClientRect();
        const html = el as HTMLImageElement;
        out[name]!.push(`${el === child ? "TOP " : ""}${el.tagName}${el.className && typeof el.className === "string" ? "." + el.className.split(" ")[0] : ""}`
          + `${el.hasAttribute("data-sphinx-preview-wide") ? " WIDE" : ""} w=${box.width.toFixed(0)} x=${box.left.toFixed(0)}`
          + `${el.matches("pre, table") ? ` cs=${el.clientWidth}/${el.scrollWidth}` : ""}${el.tagName === "IMG" ? ` natural=${html.naturalWidth}` : ""}`);
      }
    }
    return out;
  });
}

test("probe", async (t) => {
  const markdown = CASES.flatMap(([name, lines]) => [`## ${name}`, "", ...lines, ""]).join("\n");
  const root = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), "cr511-probe-")));
  const workspace = path.join(root, "workspace");
  fs.mkdirSync(workspace, { recursive: true });
  const svg = (w: number) => `<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="20"><rect width="${w}" height="20" fill="red"/></svg>`;
  fs.writeFileSync(path.join(workspace, "wide.svg"), svg(1000));
  fs.writeFileSync(path.join(workspace, "narrow.svg"), svg(100));
  const documentPath = path.join(workspace, "probe.md");
  fs.writeFileSync(documentPath, markdown);
  fs.mkdirSync(path.join(root, "out", "bundle"), { recursive: true });
  fs.writeFileSync(path.join(root, "out", "bundle", "preview.js"), pageRuntimeScript());
  const rendered = await renderPreviewDocument({
    files: createNodeFileSystem(),
    document: { uri: { scheme: "file", fsPath: documentPath }, languageId: "markdown", version: 1, getText: () => markdown },
    workspaceRoot: workspace, requestId: 1, isWorkspaceTrusted: () => true, plantUmlEngine: "builtin",
  });
  const assets = fakeAssetResolver({ cspSource: "file:", url: (p) => pathToFileURL(p).href, extensionRoot: root });
  const bundle = { ...assets, bundleUri: (...s: string[]) => (s[0] === "preview.js" ? assets.bundleUri(...s) : undefined) };
  const context = {
    activeSourceFile: documentPath, documentId: pathToFileURL(documentPath).toString(), revision: 1, requestId: 1,
    previewThemeMode: "current" as const,
    initialViewPreferences: { zoom: 100, widthMode: "margin" as const, toc: false, incomingChanges: true, comments: true, toolbarCollapsed: true },
  };
  const built = await buildPreviewWebviewHtml(createNodeFileSystem(), rendered.html, workspace, bundle, context, workspace,
    undefined, 688 + 156, true, [], "builtin");
  const pagePath = path.join(root, "page.html");
  fs.writeFileSync(pagePath, built.html);
  const browser = await chromium.launch();
  try {
    const page = await browser.newPage({ viewport: { width: 1100, height: 1400 } });
    await page.addInitScript(() => {
      (window as unknown as { acquireVsCodeApi: () => unknown }).acquireVsCodeApi = () => ({
        getState: () => ({}), setState: () => undefined, postMessage: () => undefined,
      });
    });
    await page.goto(pathToFileURL(pagePath).href);
    await page.waitForFunction(() => document.documentElement.getAttribute("data-sphinx-preview-width") === "margin");
    const margin = await read(page);
    await page.evaluate(() => document.querySelector<HTMLElement>("[data-sphinx-preview-width-toggle]")!.click());
    await page.waitForFunction(() => document.documentElement.getAttribute("data-sphinx-preview-width") === "auto");
    const auto = await read(page);
    for (const [name] of CASES) {
      const same = JSON.stringify(margin[name]) === JSON.stringify(auto[name]);
      t.diagnostic(`${name}: ${same ? "SAME as Margin" : "DIFFERS from Margin"} (${auto[name]?.length} elements)`);
      t.diagnostic(`  margin: ${margin[name]?.join(" | ")}`);
      if (!same) t.diagnostic(`  auto:   ${auto[name]?.join(" | ")}`);
    }
  } finally {
    await browser.close();
    fs.rmSync(root, { recursive: true, force: true });
  }
});
