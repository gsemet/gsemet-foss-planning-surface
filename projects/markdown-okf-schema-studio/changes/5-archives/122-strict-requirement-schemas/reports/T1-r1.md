## Script (/private/tmp/claude-502/cr122-driver/coder-r1/apply.py, run from the worktree root)

```python
#!/usr/bin/env python3
"""CR 122 mechanical change. Run from the worktree root: python3 apply.py"""
import re, subprocess, pathlib
root = pathlib.Path(".")
tiers = root / "requirements/tiers"

# 1. schemas
base = tiers / "_schema/base.schema.yaml"
t = base.read_text()
link = ("    items:\n      type: string\n      x-okf-link:\n        resolution: filename-stem\n"
        "        syntax: plain\n        id_marker: id\n        tooltip_shows_fields: [description]\n"
        "        preview_shows_fields: [description]\n")
blocks = (
 "  superseded_by:\n    type: array\n" + link +
 "    description: Requirement IDs that replace this one; required when the lifecycle\n"
 "      is superseded and forbidden otherwise.\n"
 "  supersedes:\n    type: array\n" + link +
 "    description: Generated IDs of the requirements this one replaces.\n"
 "  relates_to:\n    type: array\n" + link +
 "    description: Requirement IDs this one is laterally related to; carries no\n"
 "      derivation, supersession or coverage meaning.\n")
anchor = "  external_id:\n"
assert t.count(anchor) == 1 and "superseded_by:" not in t.split("required:")[0].split("properties:")[1]
t = t.replace(anchor, blocks + anchor)
assert t.count("additionalProperties: true\n") == 1
t = t.replace("additionalProperties: true\n", "")
base.write_text(t)

for name in ("strs", "swrs"):
    p = tiers / f"_schema/{name}.schema.yaml"
    s = p.read_text()
    # insert top-level key just before 'examples:' (after the allOf block)
    assert s.count("\nexamples:\n") == 1
    s = s.replace("\nexamples:\n", "\nunevaluatedProperties: false\nexamples:\n")
    p.write_text(s)

# 2. requirement files: scalar superseded_by -> list; StRS exemption removal
fm = re.compile(r"\A---\n(.*?)\n---\n", re.S)
for f in sorted(tiers.rglob("*.md")):
    if f.name in ("index.md", "log.md"):
        continue
    text = f.read_text()
    m = fm.match(text)
    if not m:
        continue
    head = m.group(1)
    new = re.sub(r"^superseded_by: (?![\[\s])(\S+)$", r"superseded_by:\n- \1", head, flags=re.M)
    if f.is_relative_to(tiers / "strs"):
        new = re.sub(r"^(annotation_exemption|exemption_reason):.*\n?", "", new + "\n", flags=re.M).rstrip("\n")
    if new != head:
        f.write_text("---\n" + new + "\n---\n" + text[m.end():])

# 2b. config: generated supersedes
cfg = root / "requirements/config.yml"
c = cfg.read_text()
old = "generated_fields: [derived_by, implemented_in_files"
if "supersedes" not in c:
    assert old in c
    cfg.write_text(c.replace(old, "generated_fields: [derived_by, supersedes, implemented_in_files"))

# 3. regenerate coverage (adds `supersedes`)
subprocess.run(["uv", "run", "--group", "requirements", "okfreq", "update-coverage", "requirements"], check=True)
```
